```bash
  -run '^(TestPasswordlessLoginToken_|TestPasswordlessLogin_|TestHandleLogin_LocalSessionRegression)' \
  -count=1 -v

# 聚焦竞态检测
go test ./model ./controller \
  -run '^(TestPasswordlessLoginToken_|TestPasswordlessLogin_|TestHandleLogin_LocalSessionRegression)' \
  -race -count=1

# 聚焦覆盖率
go test ./model ./controller \
  -run '^(TestPasswordlessLoginToken_|TestPasswordlessLogin_|TestHandleLogin_LocalSessionRegression)' \
  -count=1 -coverprofile=coverage-passwordless.out
go tool cover -func=coverage-passwordless.out

# 项目全量竞态测试
go test ./... -v -race
```
